const $ = id => document.getElementById(id);

const PENDING_STATUSES = new Set(['DETECTED', 'ASSESSED', 'REVIEW_PENDING', 'review']);
const APPROACH_NOTE = 'Preparar un acercamiento humano, respetuoso y relacionado con lo expresado en la conversación.';

let conversations = [];
let candidates = [];
let selectedId = null;
let working = false;

async function api(url, options = {}) {
  const response = await fetch(url, {
    headers: {'Content-Type': 'application/json'},
    ...options
  });
  if (!response.ok) {
    let data = {};
    try { data = await response.json(); } catch {}
    throw new Error(data.detail || `No se pudo completar la acción (${response.status}).`);
  }
  return response.status === 204 ? null : response.json();
}

function candidateFor(conversationId) {
  return candidates.find(item => item.origin_conversation_id === conversationId) || null;
}

function isPending(conversation) {
  return !candidateFor(conversation.id) && PENDING_STATUSES.has(conversation.status);
}

function sourceName(source) {
  const names = {
    reddit: 'Reddit',
    hackernews: 'Hacker News',
    github: 'GitHub',
    polymarket: 'Polymarket',
    youtube: 'YouTube',
    facebook: 'Facebook',
    instagram: 'Instagram',
    linkedin: 'LinkedIn'
  };
  return names[String(source || '').toLowerCase()] || source || 'una fuente pública';
}

function formatDate(value) {
  if (!value) return '';
  const date = new Date(value);
  if (Number.isNaN(date.getTime())) return '';
  return date.toLocaleDateString('es-AR', {
    day: 'numeric',
    month: 'long',
    year: 'numeric'
  });
}

function naturalState(conversation) {
  if (candidateFor(conversation.id)) return 'En acercamiento';
  const labels = {
    OBSERVING: 'Para más adelante',
    DISCARDED: 'No corresponde',
    DO_NOT_CONTACT: 'No contactar',
    REVIEWED: 'Revisada'
  };
  return labels[conversation.status] || 'Sin revisar';
}

function selectedConversation() {
  return conversations.find(item => item.id === selectedId) || null;
}

function showToast(message, isError = false) {
  const toast = $('toast');
  toast.textContent = message;
  toast.className = `toast${isError ? ' error' : ''}`;
  window.clearTimeout(showToast.timeoutId);
  showToast.timeoutId = window.setTimeout(() => {
    toast.className = 'toast hidden';
  }, 3200);
}

function setWorking(value) {
  working = value;
  ['discardButton', 'laterButton', 'approachButton', 'previousButton', 'nextButton', 'refreshButton']
    .forEach(id => { $(id).disabled = value; });
}

async function loadWorkspace({keepSelection = true} = {}) {
  try {
    await api('/health');
    $('healthBadge').textContent = 'Conectado';
    $('healthBadge').classList.remove('offline');

    const currentId = keepSelection ? selectedId : null;
    [conversations, candidates] = await Promise.all([
      api('/api/conversations'),
      api('/api/discovery-candidates')
    ]);

    if (currentId && conversations.some(item => item.id === currentId)) {
      selectedId = currentId;
    } else {
      const firstPending = conversations.find(isPending);
      selectedId = firstPending?.id || conversations[0]?.id || null;
    }
    render();
  } catch (error) {
    $('healthBadge').textContent = 'Sin conexión';
    $('healthBadge').classList.add('offline');
    showToast(error.message, true);
  }
}

function render() {
  const conversation = selectedConversation();
  const hasConversations = Boolean(conversation);
  $('emptyState').classList.toggle('hidden', hasConversations);
  $('conversationView').classList.toggle('hidden', !hasConversations);
  if (!conversation) return;

  const index = conversations.findIndex(item => item.id === conversation.id);
  $('progressText').textContent = `Conversación ${index + 1} de ${conversations.length}`;
  $('previousButton').disabled = working || index <= 0;
  $('nextButton').disabled = working || index >= conversations.length - 1;

  $('sourceText').textContent = `Encontrada en ${sourceName(conversation.source)}`;
  const author = conversation.author_name ? `Publicada por ${conversation.author_name}` : 'Autor no identificado';
  const date = formatDate(conversation.published_at);
  $('authorDateText').textContent = date ? `${author} · ${date}` : author;

  $('reviewState').textContent = naturalState(conversation);
  $('conversationTitle').textContent = conversation.title || 'Conversación sin título';
  $('conversationText').textContent = conversation.text || '';
  $('originalLink').href = conversation.conversation_url;

  const context = String(conversation.context || '').trim();
  const text = String(conversation.text || '').trim();
  const showContext = Boolean(context && context !== text);
  $('contextBlock').classList.toggle('hidden', !showContext);
  $('conversationContext').textContent = showContext ? context : '';

  const pending = isPending(conversation);
  $('decisionSection').classList.toggle('hidden', !pending);
  $('reviewedNotice').classList.toggle('hidden', pending);
  if (!pending) {
    $('reviewedText').textContent = `Esta conversación ya fue revisada: ${naturalState(conversation)}.`;
  }

  const pendingCount = conversations.filter(isPending).length;
  $('completionNotice').classList.toggle('hidden', pendingCount !== 0);
}

function moveSelection(step) {
  if (working) return;
  const conversation = selectedConversation();
  if (!conversation) return;
  const index = conversations.findIndex(item => item.id === conversation.id);
  const target = conversations[index + step];
  if (target) {
    selectedId = target.id;
    render();
    window.scrollTo({top: 0, behavior: 'smooth'});
  }
}

function selectNextPendingAfter(conversationId) {
  const start = conversations.findIndex(item => item.id === conversationId);
  const ordered = [
    ...conversations.slice(start + 1),
    ...conversations.slice(0, Math.max(start, 0))
  ];
  const next = ordered.find(isPending);
  selectedId = next?.id || conversationId;
  render();
  window.scrollTo({top: 0, behavior: 'smooth'});
}

async function saveDecision(decision, successMessage) {
  if (working) return;
  const conversation = selectedConversation();
  if (!conversation || !isPending(conversation)) return;

  setWorking(true);
  try {
    await api(`/api/conversations/${conversation.id}/reviews`, {
      method: 'POST',
      body: JSON.stringify({
        decision,
        edited_response: null,
        reviewer_notes: null,
        reviewer_identity: null
      })
    });
    showToast(successMessage);
    const decidedId = conversation.id;
    await loadWorkspace({keepSelection: true});
    selectNextPendingAfter(decidedId);
  } catch (error) {
    showToast(error.message, true);
  } finally {
    setWorking(false);
    render();
  }
}

async function completedAssessment(conversationId) {
  const history = await api(`/api/conversations/${conversationId}/assessments/v3`);
  const existing = [...history].reverse().find(item => item.assessment_status === 'COMPLETED');
  if (existing) return existing;
  return api(`/api/conversations/${conversationId}/assessments/v3`, {method: 'POST'});
}

function requestReviewer() {
  const stored = localStorage.getItem('radar-reviewer');
  if (stored) {
    approveApproach(stored);
    return;
  }
  $('reviewerName').value = '';
  $('reviewerDialog').showModal();
  window.setTimeout(() => $('reviewerName').focus(), 0);
}

async function approveApproach(reviewerIdentity) {
  if (working) return;
  const conversation = selectedConversation();
  if (!conversation || !isPending(conversation)) return;

  setWorking(true);
  try {
    const assessment = await completedAssessment(conversation.id);
    if (assessment.assessment_status !== 'COMPLETED') {
      throw new Error('No pudimos analizar esta conversación. Quedó sin cambios para que puedas revisarla más tarde.');
    }

    await api(`/api/conversations/${conversation.id}/reviews`, {
      method: 'POST',
      body: JSON.stringify({
        decision: 'APPROVE_DISCOVERY_CONTACT',
        edited_response: APPROACH_NOTE,
        reviewer_notes: null,
        reviewer_identity: reviewerIdentity
      })
    });

    localStorage.setItem('radar-reviewer', reviewerIdentity);
    showToast('Pasó a la etapa de acercamiento');
    const decidedId = conversation.id;
    await loadWorkspace({keepSelection: true});
    selectNextPendingAfter(decidedId);
  } catch (error) {
    showToast(error.message, true);
  } finally {
    setWorking(false);
    render();
  }
}

$('previousButton').onclick = () => moveSelection(-1);
$('nextButton').onclick = () => moveSelection(1);
$('refreshButton').onclick = () => loadWorkspace({keepSelection: true});
$('discardButton').onclick = () => saveDecision('DISCARD', 'Conversación descartada');
$('laterButton').onclick = () => saveDecision('KEEP_OBSERVING', 'Guardada para revisar más adelante');
$('approachButton').onclick = requestReviewer;

$('reviewerForm').onsubmit = event => {
  event.preventDefault();
  const reviewerIdentity = $('reviewerName').value.trim();
  if (!reviewerIdentity) return;
  $('reviewerDialog').close();
  approveApproach(reviewerIdentity);
};

$('cancelReviewerButton').onclick = () => $('reviewerDialog').close();

window.addEventListener('keydown', event => {
  if ($('reviewerDialog').open) return;
  if (event.key === 'ArrowLeft') moveSelection(-1);
  if (event.key === 'ArrowRight') moveSelection(1);
});

loadWorkspace({keepSelection: false});
